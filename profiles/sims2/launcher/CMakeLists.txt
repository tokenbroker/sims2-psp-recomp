cmake_minimum_required(VERSION 3.21)
project(Sims2PSPLauncher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(sims2_launcher WIN32 ../host/desktop_main.cpp ../host/display/host_settings.cpp)
target_include_directories(sims2_launcher PRIVATE ../host)
target_compile_definitions(sims2_launcher PRIVATE SIMS2_RUNTIME_SUBDIR="work/install")
target_link_libraries(sims2_launcher PRIVATE user32 shell32)
